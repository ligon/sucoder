#+begin_example
python3 - <<'PY'
import hashlib, json, subprocess
from pathlib import Path
roots = {"SuCoder": Path.cwd(),
         "LSMS_Library": Path("/home/ligon/Projects/LSMS_Library")}
data = json.loads(Path("docs/dream/corpus.json").read_text())
for corpus in data["corpora"]:
    total = 0
    for unit in corpus["sources"]:
        obj = corpus["commit"] + ":" + unit["path"]
        root = str(roots[corpus["repository"]])
        raw = subprocess.check_output(["git", "-C", root, "show", obj])
        blob = subprocess.check_output(["git", "-C", root, "rev-parse", obj],
                                       text=True).strip()
        lines = raw.splitlines(keepends=True)
        first, last = unit["lines"]
        selected = b"".join(lines[first - 1:last])
        assert blob == unit["git_blob"]
        assert hashlib.sha256(selected).hexdigest() == unit["sha256"]
        assert len(selected) == unit["bytes"]
        assert len(b"".join(lines[:first - 1])) == unit["byte_start"]
        assert len(b"".join(lines[:last])) == unit["byte_end"]
        total += len(selected)
    assert total == corpus["selected_bytes"]
    print(corpus["repository"], len(corpus["sources"]), total)
PY
#+end_example
